\begin{abstract}
Conceptual skeleton of the GPTFT program: the spacetime axioms, topological field theories as finite algebraic data, the category of GPTs as $\mathbf{Vec}$ plus positivity, the $1$d case as dualizability, and the $2$d case as a commutative Frobenius algebra whose finite data are positive GPT data. Existence of a $2$d theory reduces to an intrinsic condition on the state space --- an \emph{operational Frobenius structure}, whose physically central case is a \emph{copyable maximal frame} of perfectly distinguishable states. \emph{Spectral mountability} (every frame copyable, coherently) derives the dagger-compact postulates of the Selby--Scandolo--Coecke reconstruction, making spacetime-mountability a candidate reconstruction axiom.
\end{abstract}

\section{The spacetime axioms}\label{sec:intro}

\begin{itemize}
\item A field theory assigns: space $\to$ system, spacetime $\to$ process.
\item Locality is the compatibility of these assignments:
\[
Z(M'\circ M)=Z(M')\circ Z(M),\qquad
Z(\Sigma\sqcup\Sigma')=Z(\Sigma)\otimes Z(\Sigma').
\]
Gluing $\to$ composition (temporal locality); disjoint union $\to$ joint system (spatial locality).
\item Axioms (Atiyah--Segal \cite{Atiyah1988,Segal2004}): a field theory is a symmetric monoidal functor
\[
Z:\mathbf{Bord}_n\longrightarrow\mathcal C .
\]
\item The source is a choice of what spacetime is: in a \emph{topological} theory, bordisms up to diffeomorphism --- no metric, no local excitations.
\item The target is a choice of what physics is: $\mathbf{Vec}$, $\mathbf{Hilb}$ encode the linear skeleton of quantum theory; a \emph{GPT} target encodes directly the operational, probabilistic content \cite{Hardy2001,Barrett2007,Mueller2021,Plavala2023}.
\item \textbf{GPTFT} $:=$ TFT valued in the category of GPTs. Question: \emph{which probabilistic theories admit one?}
\end{itemize}

\section{Topological field theories}\label{sec:tft}

\begin{itemize}
\item $\mathbf{Bord}_n$: objects $=$ closed oriented $(n-1)$-manifolds; morphisms $=$ compact oriented $n$-bordisms up to diffeomorphism rel boundary; composition $=$ gluing; $(\sqcup,\varnothing)$ the monoidal structure.
\item Diffeomorphism invariance $\Rightarrow$ every manifold decomposes into finitely many elementary pieces $\Rightarrow$ a TFT is \emph{finite algebraic data} in $\mathcal C$: values on generators, subject to the relations identifying different decompositions of the same manifold.
\item $d=1$: generators interval, cup, cap; relations the snakes; a theory $=$ a \emph{dualizable object} of $\mathcal C$ (cobordism hypothesis \cite{Lurie2009}).
\item $d=2$: generators disk, pants, copants, reversed disk; a theory $=$ a \emph{commutative Frobenius algebra} in $\mathcal C$ \cite{Dijkgraaf1989,Abrams1996,Kock2004}.
\item Closed manifolds $\to$ endomorphisms of $\mathbf 1$, i.e.\ scalars: the \emph{invariants} of the theory.
\end{itemize}

\section{The category of GPTs: Vec plus positivity}\label{sec:gpt}

$\mathbf{Vec}$ (finite-dimensional real): all linear maps, monoidal product $\otimes$, compact closed --- duals, $\operatorname{ev}$, $\operatorname{coev}$ canonical, snakes automatic. GPT is the same data with one positivity condition per layer:
\[
\begin{aligned}
V\ &\rightsquigarrow\ (V,\,V^+,\,u),\\[2pt]
f\ &\rightsquigarrow\ f\ \ \text{with}\ \ f(V_A^+)\subseteq V_B^+,\\[2pt]
V_A\otimes V_B\ &\rightsquigarrow\ V_{AB}^+\in[\otimes_{\min},\otimes_{\max}],\quad u_A\otimes u_B.
\end{aligned}
\]

\begin{itemize}
\item \emph{Objects}: systems $A=(V_A,V_A^+,u_A)$; $V_A^+$ a proper cone of unnormalized states, $u_A\in V_A^*$ strictly positive on it. Normalized states $\Omega_A=u_A^{-1}(1)\cap V_A^+$; effects in the dual cone $(V_A^+)^*$; allowed effects $[0,u_A]$; measurements $(e_i)$, $\sum_ie_i=u_A$. No-restriction by default.
\item \emph{Morphisms}: cone-preserving linear maps. $\operatorname{Hom}_{\mathrm{GPT}}(A,B)$ is a \emph{cone}, not a subspace; GPT is a wide, non-full subcategory of $\mathbf{Vec}$. States and effects are arrows $\mathbf 1\to A$, $A\to\mathbf 1$; probabilities are composites $e\circ f\circ\omega\in\operatorname{End}(\mathbf 1)=\mathbb{R}$.
\item \emph{Composites}: underlying space $V_A\otimes V_B$ (tomographic locality), unit $u_A\otimes u_B$, composite cone any proper cone in $[\otimes_{\min},\otimes_{\max}]$. The interval collapses iff one factor is classical \cite{ALPP2021}; the quantum cone sits strictly between.
\item \emph{Departure from $\mathbf{Vec}$}: duals survive on objects ($X^\vee$ swaps state and effect cones onto $V^*$), but $\operatorname{ev}$ and $\operatorname{coev}$ are morphisms only if \emph{positive}. Compact closure --- automatic in $\mathbf{Vec}$ --- becomes a contingent property of the cones.
\end{itemize}

\section{One dimension: dualizability}\label{sec:1d}

\begin{itemize}
\item $\mathbf{Bord}_1$: generated by $\operatorname{id}_+$, cup $\cup:\varnothing\to+\sqcup-$, cap $\cap:-\sqcup+\to\varnothing$, modulo the snakes; $S^1=\cap\circ\sigma\circ\cup$. A $1$d theory \emph{is} a dualizable object $X=Z(+)$ \cite{Lurie2009}.
\item In $\mathbf{Vec}$: every object dualizable, $\operatorname{coev}(1)=g:=\sum_af_a\otimes f^a$, snakes $=$ resolution of identity, $Z(S^1)=\dim_{\mathbb R}V$. Classification: one integer. Nothing to check.
\item In GPT: the dual is \emph{forced}, $X^\vee=(V^*,(V^+)^*,u^\vee)$, and dualizability of $X=(V,V^+,u)$ reduces to two memberships of the \emph{single} element $g$:
\begin{align}
\text{cup (state):}&\quad g\in V^+_{X\otimes X^\vee},\label{eq:cup}\\
\text{cap (effect):}&\quad g\in\big(V^+_{X^\vee\otimes X}\big)^{*}.\label{eq:cap}
\end{align}
The cup wants the composite cone large, the cap wants it small; min--max duality exchanges the two ends. Operationally, \eqref{eq:cup}--\eqref{eq:cap} assert exact teleportation with $g$ as both resource and measurement \cite{BBLW}.
\item Conjecture: \eqref{eq:cup}--\eqref{eq:cap} are satisfiable iff $V^+$ is \emph{weakly self-dual}. Classical: passes, $g$ separable ($\in\otimes_{\min}$). Quantum: passes, $g$ $=$ maximally entangled element after a transpose twist. Boxworld: no-go (square cone not self-dual; only finitely many twists).
\item Koecher--Vinberg \cite{KoecherVinberg} $+$ BBLW steering \cite{BBLW}: homogeneous self-dual cones are Euclidean Jordan algebras $\Rightarrow$ theories passing the $1$d test are ``close to quantum.''
\end{itemize}

A $2$d theory assigns $X:=Z(S^1)=(V,V^+,u)$ and one morphism per generator (Fig.~\ref{fig:pants}); the relations (unit, counit, (co)associativity, (co)commutativity, Frobenius) are equalities of linear maps, unchanged from $\mathbf{Vec}$. The finite data, read in GPT:

\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{lll}
bordism & algebra & GPT positivity\\
\hline
disk $\eta:\varnothing\to S^1$ & unit $1_A$ & a state: $\eta(1)\in V^+$\\
rev.\ disk $\varepsilon:S^1\to\varnothing$ & counit & an effect: $\varepsilon\in(V^+)^*$\\
pants $\mu:S^1\sqcup S^1\to S^1$ & product & $\mu\big(V^+_{XX}\big)\subseteq V^+$\\
copants $\Delta:S^1\to S^1\sqcup S^1$ & coproduct & $\Delta\big(V^+\big)\subseteq V^+_{XX}$
\end{tabular}
\end{center}

\begin{defi}[$2$d GPTFT, finite data]
A GPT system $X=(V,V^+,u)$, a composite cone $V^+_{XX}\in[\otimes_{\min},\otimes_{\max}]$, and four \emph{positive} structure maps $\eta,\varepsilon,\mu,\Delta$ as in the table, satisfying the (co)unit, (co)associativity, (co)commutativity and Frobenius relations. Equivalently: a commutative Frobenius algebra in $\mathbf{Vec}$ all of whose structure maps are positive for the chosen cones.
\end{defi}

\paragraph{Conceptual points.}
\begin{itemize}
\item \emph{Operational reading.} The unit is a preparation (a distinguished unnormalized state created by the disk); the counit is a measurement outcome; merging ($\mu$) and splitting ($\Delta$) of circles are physical transformations.
\item \emph{The cone tension returns.} $\mu$ is easiest to make positive when $V^+_{XX}$ is \emph{small} (fewer inputs to control), $\Delta$ when it is \emph{large} (more room for outputs): the $2$d incarnation of the $1$d cup/cap tension, now on genuinely independent data.
\item \emph{The $1$d problem embeds.} The derived pairing and copairing
\[
\beta:=\varepsilon\circ\mu\in\big(V^+_{XX}\big)^*,\qquad
\gamma:=\Delta(\eta(1))\in V^+_{XX}
\]
are a positive effect and a positive state, and the Frobenius $+$ (co)unit relations force the snake identities for $(\gamma,\beta)$. Hence $X$ is dualizable with $X^\vee\cong X$: the $2$d structure contains a solution of the $1$d problem \emph{with the dual identified with the object itself}.
\item \emph{Self-duality becomes necessary.} Since the dual is forced to be $X^\vee=(V^*,(V^+)^*,u^\vee)$, the isomorphism $X\cong X^\vee$ is a positive linear isomorphism $V\cong V^*$ carrying $V^+$ onto $(V^+)^*$: \emph{weak self-duality of the state cone is a necessary condition for a $2$d GPTFT}. What is conjectural in $1$d is forced in $2$d. Immediate corollary: boxworld admits no $2$d GPTFT.
\item \emph{Invariants are nonnegative.} The handle operator $h:=\mu\circ\Delta$ is a positive endomorphism of $X$, and
\[
Z(\Sigma_g)=\varepsilon\big(h^g(\eta(1))\big)\ \ge\ 0 :
\]
every closed surface evaluates to a composition of positive data --- an (unnormalized) probability. In particular $Z(S^2)=\varepsilon(\eta(1))\ge0$ and $Z(T^2)=\dim_{\mathbb R}V$.
\item \emph{Classical passes --- twice over.} Basis type: the pointwise algebra on $\mathbb{R}^N$, $\eta(1)=\sum_ie_i$, $\varepsilon(e_i)=\lambda_i>0$, $\mu(e_i\otimes e_j)=\delta_{ij}e_i$, $\Delta(e_i)=\lambda_i^{-1}e_i\otimes e_i$; one family per weight vector $\lambda$, with invariants $Z(\Sigma_g)=\sum_i\lambda_i^{1-g}$. Gauge type (no copyable frame): the nilpotent algebra $\mathbb{R}[x]/(x^2)$ on the bit cone $\operatorname{cone}\{1,x\}$ with $\varepsilon=s\,x^*$ ($Z(S^2)=0$, $Z(T^2)=2$), and group algebras $\mathbb{R}[G]$ with cone over the group elements --- $2$d lattice gauge theories mounted on classical GPTs, whose would-be frames (Fourier idempotents) are not physical states.
\item \emph{Quantum passes --- the coherent copy.} For each orthonormal basis, with $W=\sum_i|ii\rangle\langle i|$:
\[
\Delta(\rho)=W\rho W^\dagger,\quad
\mu(x)=W^\dagger xW\ \ (\text{Schur product}),\quad
\eta(1)=J=\textstyle\sum_{ij}|i\rangle\langle j|,\quad
\varepsilon=\langle\eta|\cdot|\eta\rangle .
\]
All four maps are positive for the standard quantum composite (positivity of $\mu$ \emph{is} the Schur product theorem); the derived cup is the maximally entangled $\gamma=|\Phi\rangle\langle\Phi|$ --- realizing the transpose twist conjectured in the $1$d analysis --- the handle is trivial ($h=\operatorname{id}$, the structure is special), and $Z(\Sigma_g)=N^2$ for every genus. No-cloning is respected: the \emph{trace}-marginals of $\Delta$ give dephasing; only the $\varepsilon$-marginal is lossless.
\item \emph{No-broadcasting dichotomy.} If $\varepsilon$ is strictly positive, the counit laws make $\Delta$ a universal broadcaster for the system $(V,V^+,\varepsilon)$, and the generalized no-broadcasting theorem \cite{BBLWnobroadcast} forces a simplicial cone. Nonclassical solutions therefore have \emph{boundary} counit --- in quantum, the pure effect $\langle\eta|\cdot|\eta\rangle$. Diagrammatically \cite{SSC2021}: purity of the cup $\iff$ triviality of leaks $\iff$ fully quantum.
\end{itemize}

\section{The existence condition: copyable frames}\label{sec:condition}

\paragraph{Reduction to the state space.}
All the data of a $2$d GPTFT collapse onto a product and one effect:

\begin{prop}[Operational Frobenius structure]\label{prop:ofs}
$X=(V,V^+,u)$ admits a $2$d GPTFT iff $V$ carries a commutative associative product $\star$ with unit $\eta$, and an effect $\varepsilon\in(V^+)^*$, such that:
\emph{(i)} $\eta\in V^+$ and $V^+\!\star V^+\subseteq V^+$ --- \emph{the cone is closed under the product};
\emph{(ii)} $\beta(v,w):=\varepsilon(v\star w)$ is nondegenerate and $\flat:v\mapsto\beta(v,-)$ is an order isomorphism $(V,V^+)\cong(V^*,(V^+)^*)$ --- \emph{the effect self-dualizes the cone through the product};
\emph{(iii)} the handle element $H=\sum_a b_a\star b^a$ (any basis, $\beta$-dual basis) acts positively.
The coproduct and counit are then derived, $\Delta=\mu^{\dagger}$, $\varepsilon=\eta^{\dagger}$, and the composite cone can be taken minimal: $V^+_{XX}=\overline{\operatorname{cone}}\big(\otimes_{\min}\cup\,\Delta(V^+)\big)$.
\end{prop}

\paragraph{Dagger form (under tomographic locality).}
$\beta$ induces a dagger $f^\dagger:=\flat^{-1}\!\circ f^*\circ\flat$ on the monoidal subcategory generated by $X$; condition \emph{(ii)} says precisely that this dagger \emph{preserves positivity}. A $2$d GPTFT is then a positive dagger together with a commutative $\dagger$-Frobenius algebra $(\mu,\eta)$ with $\mu,\eta$ positive --- the GPT form of the correspondence between $\dagger$-Frobenius algebras, orthonormal bases, and finite-dimensional C$^*$-algebras \cite{CPV2013,Vicary2011}. Two bilinear forms coexist and divide the labor:
\[
\beta=\varepsilon\circ\mu,
\qquad
\beta_s(v,w)=\varepsilon(v\star Tw),
\]
where $T$ is the canonical \emph{Galois involution} of the algebra (identity on the real spectrum, conjugation on the complex spectrum; in quantum, $\rho\mapsto\rho^{T}$). The copairing of $\beta$ is the positive cup --- the teleportation resource; in quantum $\gamma=|\Phi\rangle\langle\Phi|$, $\beta=\operatorname{tr}(\rho\,\sigma^{T})$. The copairing of $\beta_s$ is \emph{not} positive (in quantum, \textsc{swap}), but $\beta_s$ is a GNS-type inner product whose adjoint is the \emph{sharp dagger} (in quantum, Hilbert--Schmidt, giving the Hermitian adjoint). Cups give the transpose; the Frobenius algebra supplies the conjugation; their composite is the dagger --- $(\star,T,\varepsilon)$ is an abstract commutative $*$-algebra with a positive functional, compatible with the cone.

\paragraph{The operational condition.}

\begin{defi}[Copyable frame]
A maximal set of perfectly distinguishable states $\{\omega_i\}_{i=1}^{n}$ of $X$ is \emph{copyable} if there exist an admissible composite cone, a positive linear map $\Delta:X\to X\otimes X$, and an effect $\varepsilon\in(V^+)^*$ with:
\emph{(C1) copying:} $\Delta(\omega_i)=\omega_i\otimes\omega_i$;
\emph{(C2) losslessness:} $\varepsilon(\omega_i)=1$ and $(\varepsilon\otimes\operatorname{id})\Delta=\operatorname{id}=(\operatorname{id}\otimes\varepsilon)\Delta$;
\emph{(C3) consistency:} $\Delta$ is coassociative and cocommutative;
\emph{(C4) mergeability:} the copairing dual to the induced $\beta$ equals $\Delta(\eta)$ for a state $\eta\in V^+$, and the induced merge $\mu$ is positive.
\end{defi}

\begin{defi}[Spectrally mountable]
$X$ is \emph{spectrally mountable} if \emph{every} maximal set of perfectly distinguishable states is copyable, and the daggers induced by the twisted forms $\beta_s$ of the resulting structures coincide (\emph{coherence}).
\end{defi}

\begin{prop}[Existence]\label{prop:exist}
A single copyable frame yields a $2$d GPTFT of \emph{basis type}: the frame is the positive real spectrum of the algebra, and the twisted dagger is sharp at it. Every basis-type theory arises this way. The inclusion in all $2$d GPTFTs is strict: \emph{gauge-type} solutions exist with no copyable frame (nilpotent and group-algebra structures above). Every $2$d GPTFT has at least one real character (Perron--Frobenius applied to the commuting positive family $L_v=v\star(-)$).
\end{prop}

\begin{prop}[Alignment with the SSC reconstruction \cite{SSC2021}]\label{prop:ssc}
If $X$ is spectrally mountable in a tomographically local GPT, then Postulates 1--4 of \cite{SSC2021} hold for the process theory generated by $X$: \emph{(1)} process theory and \emph{(2)} finite local classical interface are the ambient framework; \emph{(3)} cups \& caps are $\gamma=\Delta(\eta)$ and $\beta=\varepsilon\circ\mu$; \emph{(4)} the common twisted dagger is sharp at every maximal testable preparation, since every frame is copyable. By the lemmas of \cite{SSC2021}, $X$ is then \emph{spectral}. Their Postulate 5 (essentially unique symmetric purification $\Rightarrow$ homogeneity $\Rightarrow$ Koecher--Vinberg $\Rightarrow$ Euclidean Jordan algebras) is the remaining distance to quantum $\oplus$ classical theory.
\end{prop}

Quantum theory is spectrally mountable: one coherent copier per orthonormal basis, and all twisted forms collapse to the basis-independent Hilbert--Schmidt inner product, whose adjoint is the Hermitian dagger. Classical theory is spectrally mountable through the pointwise copiers. Boxworld fails already at the derived $1$d layer.

\paragraph{The ladder.}
Each condition strictly refines the last (as classes of GPTs):
\[
\{\text{teleportable}\}\supseteq\{\text{mountable}\}\supseteq\{\text{copyable frame}\}\supseteq\{\text{spectrally mountable}\}\supseteq\{\text{quantum}\oplus\text{classical}\},
\]
where \emph{teleportable} $=$ admits a $1$d GPTFT (cups \& caps; weak self-duality), \emph{mountable} $=$ admits some $2$d GPTFT (operational Frobenius structure; at least one real character; nonnegative invariants), \emph{copyable frame} $=$ admits a basis-type theory, and \emph{spectrally mountable} as above ($\Rightarrow$ SSC Postulates 1--4 $\Rightarrow$ spectrality). The last inclusion is conjectured to be an equality up to the Jordan-algebraic loose ends.

\paragraph{Open problems.}
\emph{(1) Coherence:} is the common-dagger condition automatic when every frame is copyable?
\emph{(2) Converse of Prop.~\ref{prop:ssc}:} can the coherent copier be constructed from a sharp dagger --- the ``spiders'' question raised in the outlook of \cite{SSC2021}?
\emph{(3) Spin factors:} are the $d$-ball systems ($d\neq3$) mountable? Decisive for whether $2$d mountability alone cuts the Jordan family to $\mathbb{C}$-quantum $\oplus$ classical.
\emph{(4)} The TFT origin of symmetric purification (SSC Postulate 5).
\emph{(5)} The non-tomographically-local ($C\neq0$, fermionic) case.
